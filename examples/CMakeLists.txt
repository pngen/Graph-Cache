# Runnable examples. CPU examples are always built; CUDA example gated on CUDA.
function(gc_example name source)
  add_executable(example_${name} ${source})
  target_link_libraries(example_${name} PRIVATE graphcache)
  target_include_directories(example_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  if(GRAPHCACHE_HAS_CUDA)
    target_compile_definitions(example_${name} PRIVATE GC_HAS_CUDA=1)
  endif()
  add_test(NAME example_${name} COMMAND example_${name})
  set_tests_properties(example_${name} PROPERTIES TIMEOUT 0)
endfunction()
